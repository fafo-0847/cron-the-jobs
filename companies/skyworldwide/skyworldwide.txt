import hashlib
import json
import logging
import datetime
from pathlib import Path
from typing import Any

from playwright.async_api import async_playwright

logging.basicConfig(level=logging.INFO)
logger = logging.getLogger("skyworldwide")

sourceUrl = "https://jumpit.saramin.co.kr/company/MjIwODg2ODQwOA==?company_nm=%EC%8A%A4%EC%B9%B4%EC%9D%B4%EC%9B%94%EB%93%9C%EC%99%80%EC%9D%B4%EB%93%9C"
baseUrl = "https://jumpit.saramin.co.kr"
companyDir = Path(__file__).parent
snapshotPath = companyDir / "skyworldwide.json"


def generateJobId(title: str, stableId: str) -> str:
	hasStableId = stableId != ""
	if hasStableId:
		raw = title + stableId
	else:
		raw = title
	jobId = hashlib.md5(raw.encode()).hexdigest()
	return jobId


def extractStableId(href: str) -> str:
	segments = href.strip("/").split("/")
	hasSegments = len(segments) > 0
	if hasSegments:
		stableId = segments[-1]
	else:
		stableId = ""
	return stableId


def normalizeText(rawText) -> str:
	isNone = rawText is None
	if isNone:
		return ""
	normalized = rawText.replace("\xa0", " ").strip()
	return normalized


async def fetchState(url: str = sourceUrl) -> dict[str, Any]:
	jobs = []
	try:
		async with async_playwright() as playwright:
			browser = await playwright.chromium.launch()
			page = await browser.new_page()
			await page.goto(url, wait_until="domcontentloaded", timeout=30000)
			await page.wait_for_selector("a[href*='/position/']", timeout=20000)
			positionLinkLocator = page.locator("a[href*='/position/']")
			positionLinkCount = await positionLinkLocator.count()
			index = 0
			while index < positionLinkCount:
				currentLink = positionLinkLocator.nth(index)
				titleLocator = currentLink.locator("h3")
				titleCount = await titleLocator.count()
				hasTitle = titleCount > 0
				if hasTitle:
					titleText = await titleLocator.first.inner_text()
					title = normalizeText(titleText)
					hasNonEmptyTitle = title != ""
					if hasNonEmptyTitle:
						href = await currentLink.get_attribute("href")
						hasHref = href is not None
						if not hasHref:
							href = ""
						techStackLocator = currentLink.locator("ul li")
						techStackCount = await techStackLocator.count()
						techStackValues = []
						techIndex = 0
						while techIndex < techStackCount:
							techText = await techStackLocator.nth(techIndex).inner_text()
							techValue = normalizeText(techText).lstrip("·").strip()
							techStackValues.append(techValue)
							techIndex = techIndex + 1
						techStack = ", ".join(techStackValues)
						deadlineBadgeLocator = currentLink.locator("label span")
						deadlineBadgeCount = await deadlineBadgeLocator.count()
						hasDeadlineBadge = deadlineBadgeCount > 0
						if hasDeadlineBadge:
							deadlineBadgeText = await deadlineBadgeLocator.first.inner_text()
							daysRemaining = normalizeText(deadlineBadgeText)
						else:
							daysRemaining = ""
						stableId = extractStableId(href)
						jobId = generateJobId(title, stableId)
						link = baseUrl + href
						job = {
							"id": jobId,
							"title": title,
							"link": link,
							"location": "",
							"department": "",
							"jobType": "",
							"techStack": techStack,
							"daysRemaining": daysRemaining,
							"postedAt": "",
							"state": "open",
						}
						jobs.append(job)
				index = index + 1
			await browser.close()
	except Exception as error:
		logger.exception("skyworldwide: fetchState failed: %s", error)
	state = {"jobs": jobs, "state": "open"}
	return state


def loadExistingSnapshot() -> dict[str, Any]:
	try:
		fileExists = snapshotPath.exists()
		if not fileExists:
			return {"sourceUrl": sourceUrl, "fetchedAt": "", "state": "open", "jobs": []}
		rawText = snapshotPath.read_text(encoding="utf-8")
		snapshot = json.loads(rawText)
		return snapshot
	except Exception as error:
		logger.exception("skyworldwide: loadExistingSnapshot failed: %s", error)
		return {"sourceUrl": sourceUrl, "fetchedAt": "", "state": "open", "jobs": []}


def buildSnapshot(state: dict, existingSnapshot: dict) -> dict:
	oldJobs = existingSnapshot.get("jobs", [])
	newJobs = state.get("jobs", [])
	oldJobCount = len(oldJobs)
	newJobCount = len(newJobs)
	suspiciousDrop = oldJobCount > 0 and newJobCount == 0
	if suspiciousDrop:
		logger.warning("skyworldwide: suspicious zero-job scrape, keeping old snapshot. Old=%d New=%d", oldJobCount, newJobCount)
		return dict(existingSnapshot)
	fetchedAt = datetime.datetime.now(datetime.timezone.utc).isoformat()
	snapshot = {
		"sourceUrl": sourceUrl,
		"fetchedAt": fetchedAt,
		"state": state.get("state", "open"),
		"jobs": newJobs,
	}
	return snapshot


def writeSnapshot(snapshot: dict) -> None:
	try:
		snapshotText = json.dumps(snapshot, indent=2, ensure_ascii=False)
		snapshotPath.write_text(snapshotText, encoding="utf-8")
	except Exception as error:
		logger.exception("skyworldwide: writeSnapshot failed: %s", error)


async def updateSnapshot(url: str = sourceUrl) -> dict:
	existingSnapshot = loadExistingSnapshot()
	state = await fetchState(url)
	snapshot = buildSnapshot(state, existingSnapshot)
	writeSnapshot(snapshot)
	return snapshot


async def runMain() -> None:
	snapshot = await updateSnapshot()
	jobCount = len(snapshot.get("jobs", []))
	logger.info("skyworldwide: run complete, %d jobs in snapshot", jobCount)


if __name__ == "__main__":
	import asyncio
	asyncio.run(runMain())
